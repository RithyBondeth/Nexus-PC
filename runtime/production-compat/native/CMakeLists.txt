cmake_minimum_required(VERSION 3.22.1)
project(NexusRuntimeNative C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384")
if(NOT NEXUS_UPSTREAM_CPP)
  message(FATAL_ERROR "NEXUS_UPSTREAM_CPP is required")
endif()
set(SRC ${NEXUS_UPSTREAM_CPP})
add_subdirectory(${SRC}/winlator winlator)
add_subdirectory(${SRC}/libadrenotools libadrenotools)
add_subdirectory(${SRC}/vortekrenderer vortekrenderer)
add_subdirectory(${SRC}/gladiorenderer gladiorenderer)
add_subdirectory(${SRC}/virglrenderer virglrenderer)
